glob { result }
proc {
    setone ( x ) { local { } x = 1 }
    settwo ( x y ) { local { } x = 1; y = 2 }
    setthree ( x y z ) { local { } x = 1; y = 2; z = 3 }
}
func {
    addone ( a ) { local { temp } temp = a; return temp }
    addtwo ( a b ) { local { temp } temp = ( a plus b ); return temp }
    addthree ( a b c ) { local { temp temp2 } temp = ( a plus b ); temp2 = ( temp plus c ); return temp2 }
}
main {
    var { a b c }
    a = 0;
    b = 0;
    c = 0;
    
    setone ( a );
    print "after setone";
    
    settwo ( a b );
    print "after settwo";
    
    setthree ( a b c );
    print "after setthree";
    
    result = addone ( 5 );
    print result;
    
    result = addtwo ( 3 7 );
    print result;
    
    result = addthree ( 2 3 4 );
    print result;
    
    halt
}
